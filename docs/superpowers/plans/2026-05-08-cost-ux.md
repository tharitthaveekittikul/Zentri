# Cost UX Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show users the estimated cost before every one-shot LLM call (confirmation dialog with primary + secondary currency) and a persistent cost badge in chat. No LLM call fires without explicit user confirmation.

**Architecture:** A shared `ConfirmLLMDialog` component is reused across all one-shot features. A `useLLMCost` hook estimates cost client-side from token count × model pricing. A `ChatCostBadge` component shows `~฿X / $Y per message` in the chat header. Currency conversion rates come from the existing exchange rate endpoint.

**Dependencies:** Plans 1–3 must be complete — all LLM trigger points must exist before we wire dialogs to them.

**Tech Stack:** Next.js, TypeScript, React, shadcn/ui

```typescript
// Cost per 1M tokens in USD. Update when providers change pricing.
export interface ModelPricing {
  inputPer1M: number   // USD per 1M input tokens
  outputPer1M: number  // USD per 1M output tokens
  label: string
}

export const MODEL_PRICING: Record<string, ModelPricing> = {
  // Anthropic
  "claude-haiku-4-5-20251001":    { inputPer1M: 0.80,  outputPer1M: 4.00,  label: "Claude Haiku 4.5" },
  "claude-sonnet-4-6":            { inputPer1M: 3.00,  outputPer1M: 15.00, label: "Claude Sonnet 4.6" },
  "claude-opus-4-7":              { inputPer1M: 15.00, outputPer1M: 75.00, label: "Claude Opus 4.7" },
  // OpenAI
  "gpt-4o-mini":                  { inputPer1M: 0.15,  outputPer1M: 0.60,  label: "GPT-4o mini" },
  "gpt-4o":                       { inputPer1M: 2.50,  outputPer1M: 10.00, label: "GPT-4o" },
  // Gemini
  "gemini-flash":                 { inputPer1M: 0.075, outputPer1M: 0.30,  label: "Gemini Flash" },
  "gemini-pro":                   { inputPer1M: 1.25,  outputPer1M: 5.00,  label: "Gemini Pro" },
}

export function estimateCostUsd(
  model: string,
  estimatedInputTokens: number,
  estimatedOutputTokens: number
): number {
  const pricing = MODEL_PRICING[model]
  if (!pricing) return 0
  return (
    (estimatedInputTokens / 1_000_000) * pricing.inputPer1M +
    (estimatedOutputTokens / 1_000_000) * pricing.outputPer1M
  )
}

// Rough token estimate: ~4 chars per token
export function estimateTokens(text: string): number {
  return Math.ceil(text.length / 4)
}
```

### Task 2: useLLMCost Hook

**Files:**
- Create: `frontend/hooks/useLLMCost.ts`

- [ ] **Step 1: Create hook**

Create `frontend/hooks/useLLMCost.ts`:

```typescript
import { useCallback, useEffect, useState } from "react"
import { estimateCostUsd } from "@/lib/llmPricing"

interface ExchangeRates {
  usd_to_primary: number
  usd_to_secondary: number
  primary_currency: string
  secondary_currency: string
}

interface CostEstimate {
  usd: number
  primary: number
  secondary: number
  primaryCurrency: string
  secondaryCurrency: string
  label: string   // e.g. "฿0.11 / $0.003"
}

async function fetchRates(): Promise<ExchangeRates | null> {
  try {
    const resp = await fetch("/api/settings/currencies")
    if (!resp.ok) return null
    return resp.json()
  } catch {
    return null
  }
}

export function useLLMCost(
  model: string,
  estimatedInputTokens: number,
  estimatedOutputTokens = 300
) {
  const [rates, setRates] = useState<ExchangeRates | null>(null)

  useEffect(() => {
    fetchRates().then(setRates)
  }, [])

  const estimate = useCallback((): CostEstimate => {
    const usd = estimateCostUsd(model, estimatedInputTokens, estimatedOutputTokens)
    const primaryRate = rates?.usd_to_primary ?? 35   // fallback: 35 THB/USD
    const secondaryRate = rates?.usd_to_secondary ?? 1
    const primary = usd * primaryRate
    const secondary = usd * secondaryRate
    const pCur = rates?.primary_currency ?? "THB"
    const sCur = rates?.secondary_currency ?? "USD"

    const fmt = (n: number) => n < 0.01 ? n.toFixed(4) : n.toFixed(2)
    return {
      usd,
      primary,
      secondary,
      primaryCurrency: pCur,
      secondaryCurrency: sCur,
      label: `${pCur} ${fmt(primary)} / $${fmt(usd)}`,
    }
  }, [model, estimatedInputTokens, estimatedOutputTokens, rates])

  return { estimate }
}
```

### Task 3: ConfirmLLMDialog Component

**Files:**
- Create: `frontend/components/llm/ConfirmLLMDialog.tsx`

- [ ] **Step 1: Create component**

Create `frontend/components/llm/ConfirmLLMDialog.tsx`:

```tsx
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog"
import { Button } from "@/components/ui/button"
import { Zap } from "lucide-react"
import { useLLMCost } from "@/hooks/useLLMCost"

interface Props {
  open: boolean
  onConfirm: () => void
  onCancel: () => void
  featureLabel: string       // e.g. "Portfolio Analysis"
  model: string              // model ID from feature config
  provider: string           // e.g. "Anthropic"
  estimatedInputTokens: number
}

export function ConfirmLLMDialog({
  open,
  onConfirm,
  onCancel,
  featureLabel,
  model,
  provider,
  estimatedInputTokens,
}: Props) {
  const { estimate } = useLLMCost(model, estimatedInputTokens)
  const cost = estimate()

  return (
    <Dialog open={open} onOpenChange={(o) => !o && onCancel()}>
      <DialogContent className="sm:max-w-sm">
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <Zap className="h-4 w-4" />
            Run AI Analysis?
          </DialogTitle>
          <DialogDescription>
            This will send your data to the AI provider.
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-2 text-sm">
          <div className="flex justify-between">
            <span className="text-muted-foreground">Feature</span>
            <span className="font-medium">{featureLabel}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-muted-foreground">Model</span>
            <span className="font-medium">{model}</span>
          </div>
          <div className="flex justify-between">
            <span className="text-muted-foreground">Provider</span>
            <span className="font-medium">{provider}</span>
          </div>
          <div className="flex justify-between border-t pt-2">
            <span className="text-muted-foreground">Est. cost</span>
            <span className="font-semibold text-foreground">
              {cost.primaryCurrency} {cost.primary.toFixed(2)} / ${cost.usd.toFixed(4)}
            </span>
          </div>
        </div>

        <DialogFooter>
          <Button variant="outline" onClick={onCancel}>Cancel</Button>
          <Button onClick={onConfirm}>Confirm</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```
